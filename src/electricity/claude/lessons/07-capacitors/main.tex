\telosmasthead[Follow charge through time, then prove where it went]
\lessonhead{7}{three measured runs}{isolated 3 V; adult controls energizing}
\begingroup\footnotesize

\begin{materials}
\item Switched holder with two AA cells
\item 1000 \(\mu\)F electrolytic capacitor, 10 V or higher
\item 10 k\(\Omega\) and 22 k\(\Omega\) resistors
\item Digital multimeter with insulated leads
\item Stopwatch, graph paper, removable \(+\)/\( - \) labels
\end{materials}

\begin{predict}
Imagine equal snapshots taken while an empty capacitor charges through a
resistor. Will its voltage gain be the same in every snapshot, or will the gains
shrink? Draw the rise. Beside it, draw what you expect after the battery is
removed and the capacitor is allowed to discharge through the same resistor.
\end{predict}

\lessonsection{Four states, one permitted transition at a time}

\begin{center}\vspace{-.7em}\scriptsize
\fbox{\textbf{A COLD}: source out; \(V_C<0.10\) V}
\(\longrightarrow\)
\fbox{\textbf{B CHECKED}: polarity; rating; S open}
\(\longrightarrow\)
\fbox{\textbf{C TIMING}: source in; hands away}
\(\longrightarrow\)
\fbox{\textbf{D DISCHARGE}: source out; \(R\) across \(C\)}
\(\longrightarrow\) A
\vspace{-.7em}
\end{center}

\vspace{-5pt}
\begin{teloshazard}{The circuit has only one discharge tool}
Never bridge capacitor leads with a screwdriver, wire, or probe tip. Remove the
source, then leave a resistor across the capacitor until the meter reads below
0.10 V. A small part can still heat, leak, or rupture if reversed. Stop for
warmth, swelling, odor, liquid, damage, or an unexpected reading.
\end{teloshazard}
\vspace{-5pt}

\lessonsection{Run 0 — learn the endpoints}

\begin{enumerate}
\item With the capacitor absent, measure the source:
  \(V_S=\writein{1.2cm}\) V. Remove the holder.
\item Assemble the charge path. Confirm \(V_C<0.10\) V, then the adult installs
  the holder. Ask: \emph{what sign should appear, and what would a minus sign
  tell us?}
\item Close the switch for 60 s. Record \(V_{\rm high}=\writein{1.2cm}\) V.
  Open the switch; remove the holder; use the discharge path.
\item Do not change the circuit until \(V_C=\writein{1.2cm}\) V, below 0.10 V.
\end{enumerate}

\begin{teloscard}{Calculate the landmark before comparing it with the graph}
\[
C=1000\,\mu{\rm F}=0.001\,{\rm F},\qquad
\tau=RC=(10{,}000\,\Omega)(0.001\,{\rm F})=10\,{\rm s}.
\]
For charge, \(V_C=V_S(1-e^{-t/RC})\). Thus at 10 s the prediction is
\(0.632V_S\). With measured \(V_S=\writein{1cm}\) V, the predicted 10 s reading
is \writein{1.2cm} V. The two measured readings were \writein{1cm} and
\writein{1cm} V.
\end{teloscard}

\lessonsection{Stored energy: calculate it, do not dramatize it}

For this test,
\(E=\tfrac12CV^2=\tfrac12(0.001\,{\rm F})(3.00\,{\rm V})^2=0.0045\,{\rm J}\).
Doubling voltage would multiply energy by \(2^2=4\), although capacitance did
not change. This experiment stays at 3 V and never produces a spark; its useful
observation is the measured voltage history.
\vspace{-4pt}

\begin{teloscard}{Questions that connect the curves to the circuit}
\begin{enumerate}[itemsep=.12em,topsep=.12em]
\item Why is the charge curve steepest at the beginning?
  \par\writein{.92\linewidth}
\item Why does opening the source path not make \(V_C\) instantly zero?
  \par\writein{.92\linewidth}
\item During discharge, did the voltage sign reverse or keep its sign while its
  magnitude fell? \writein{3cm}
\item What evidence distinguishes “the capacitor is empty” from “the source is
  disconnected”? \writein{5cm}
\end{enumerate}
\end{teloscard}
\vspace{-5pt}

\lessonsection{Diagnose only in State A}

\begin{tabularx}{\linewidth}{@{}p{.25\linewidth}LL@{}}
\toprule
What the meter shows & De-energized check & Correction before another run\\
\midrule
Negative sign & Probe placement and capacitor stripe & Red to labeled \(+\);
preserve capacitor polarity.\\
Near 0 V throughout charge & Switch path, clip seating, resistor bands &
Restore the complete series path; never bypass \(R\).\\
Starts above 0.10 V & Residual charge & Source out; resistor across \(C\);
wait and verify.\\
Curve too fast or slow & Actual \(R\) and printed \(C\) & Recalculate \(RC\)
from the parts actually fitted.\\
Trials disagree sharply & Timing call and loose leads & Secure with switch open;
restart both trials.\\
Warmth, swelling, odor, leak & No further test & Adult removes the holder and
retires the capacitor.\\
\bottomrule
\end{tabularx}
\vspace{-4pt}

\begin{teloshazard}{Polarity and discharge check before every change}
Open the switch. Remove the battery holder. Put the resistor across the
capacitor. Watch \(V_C\) fall below 0.10 V. Read the meter sign and magnitude;
never infer zero charge merely because the battery is absent.
\end{teloshazard}
\vspace{-5pt}

\begin{teloscard}{Final proof}
\begin{enumerate}[itemsep=.08em,topsep=.08em]
\item \textbf{Repeatability:} the two 10 k\(\Omega\) values at 10 s differed by
  \writein{1cm} V.
\item \textbf{Landmark:} mean \(V_C(10\,{\rm s})/V_S=\writein{1cm}\), compared
  with 0.632.
\item \textbf{Direction:} charge approached \writein{1cm}; discharge approached
  \writein{1cm}.
\item \textbf{Control:} changing only \(R\) multiplied the observed time by
  \writein{1cm}, so the evidence \tickbox\ supports \quad\tickbox\ challenges
  \(\tau=RC\).
\item \textbf{Safe finish:} source removed; resistor fitted;
  \(V_C=\writein{1cm}\) V \(<0.10\) V. Adult initials: \writein{1cm}.
\end{enumerate}
\end{teloscard}
