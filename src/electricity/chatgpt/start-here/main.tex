\section{Roles that make independence possible}
\begin{telosfacts}[0.22]
\telosfact{Learner}{Chooses the prediction; wires only after the power source is
disconnected; reads the diagram; records raw values; explains uncertainty.}
\telosfact{Adult facilitator}{Selects only the listed low-energy parts; inspects
cells and leads; checks meter mode and jacks; stops heat, odor, swelling, damaged
insulation, or repeated unexplained readings.}
\telosfact{Nobody in this course}{Opens an appliance; inserts anything into an
outlet; measures household power; substitutes a wall adapter; or treats “off”
as proof of deenergization.}
\end{telosfacts}

\begin{gate}{Before power is connected}
Circuit matches the diagram [\ ]\quad No direct battery short
[\ ]\quad Meter mode and lead jacks checked [\ ]\quad Adult
checks once [\ ]\quad Connect only long enough to measure [\ ]
\end{gate}
