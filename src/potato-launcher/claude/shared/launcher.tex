\newcommand{\launchersection}[5]{%
  \begin{scope}[shift={(#1,#2)},scale=#3]
    % chamber body
    \draw[pvc,pvcfill] (0,0) rectangle (#4,1.5);
    \draw[pvcdetail] (0,0.14) -- (#4,0.14);
    \draw[pvcdetail] (0,1.36) -- (#4,1.36);
    % screw-on cleanout at the breech
    \draw[pvc,pvcfill] (-0.45,-0.12) rectangle (0,1.62);
    \foreach \y in {0.1,0.35,0.6,0.85,1.1,1.35} {
      \draw[pvcdetail] (-0.45,\y) -- (-0.30,\y+0.06);}
    % reducer
    \draw[pvc,pvcfill] (#4,0) -- (#4+0.7,0.55) -- (#4+0.7,0.95) -- (#4,1.5) -- cycle;
    % barrel
    \draw[pvc,pvcfill] (#4+0.7,0.55) rectangle (#4+0.7+#5,0.95);
    \draw[pvcdetail] (#4+0.7,0.66) -- (#4+0.7+#5,0.66);
    \draw[pvcdetail] (#4+0.7,0.84) -- (#4+0.7+#5,0.84);
    % igniter, through the chamber wall
    \draw[pvc] (#4*0.42,1.5) -- (#4*0.42,2.05);
    \draw[pvc,pvcfill] (#4*0.42-0.22,2.05) rectangle (#4*0.42+0.22,2.45);
    \fill[telosink] (#4*0.42-0.06,1.62) circle (0.05);
    \fill[telosink] (#4*0.42+0.06,1.62) circle (0.05);
    \draw[telos hair] (#4*0.42-0.03,1.68) -- (#4*0.42,1.58) -- (#4*0.42+0.03,1.68);
  \end{scope}}

\newcommand{\launchernote}{%
  \begin{teloshazard}{This is a projectile weapon}
  A combustion launcher throws a half-pound potato several hundred feet at
  speeds that will kill a person. It is not a toy, it is not a firework, and the
  fact that it is made of plumbing parts changes nothing about what it does
  downrange. Every rule on the \textbf{Safety} sheet applies every time: adult
  present and in charge, muzzle discipline as strict as a rifle's, a confirmed
  clear range, eyes and ears on, and never a person, animal, building, road or
  anything you cannot afford to destroy in the direction it is pointing.
  \end{teloshazard}}

\newcommand{\pvcnote}{%
  \begin{telosframe}
  \textbf{Pressure-rated pipe only.} Schedule 40 \emph{pressure} PVC, printed
  with an NSF-PW rating and a pressure figure. Never DWV, never foam-core,
  never cellular-core, never thin-wall drain pipe --- those are not rated for
  internal pressure at all and they fail by shattering into fragments rather
  than splitting. This is the one specification in the whole project that is
  not negotiable.
  \end{telosframe}}
