\telosmasthead[Micropterus dolomieu]
\begin{speciestop}{smallmouth-bass}{\dentoncredit{The Small-Mouthed Black Bass}}
\telosfact{Tell it apart}{Bronze rather than green, with dark \emph{vertical} bars instead of one horizontal stripe, and a red or orange eye. The upper jaw stops under the eye. The dorsal fin is one continuous fin, not pinched.}
\telosfact{Size here}{12--17\,in typical; 18\,in and up is a strong fish. State record 9\,lb 1\,oz. Pound for pound the hardest-fighting fish in either lake.}
\telosfact{Spawn}{Gravel and rubble in 3--10 ft at 60--65\,\unitof{^\circ F}, a little later and deeper than largemouth. Pine Lake's 90\% gravel bottom is close to ideal smallmouth habitat.}
\telosfact{Eats}{Crayfish above all, then minnows, perch fry and insect larvae. Match the crayfish and you are most of the way there.}

\section{Where}
\begin{whenwhere}
\telosfact{Pine}{This is a smallmouth lake. Spots \textbf{2} (north-east breaks), \textbf{3} (east wall), \textbf{4} (the mid-lake bar) and \textbf{8}. The bar is the best smallmouth spot in the county on a good October morning.}
\telosfact{North}{Spots \textbf{3} and \textbf{4}. The south-end shoal complex is the equivalent of Pine's bar.}
\telosfact{Structure}{Hard bottom with a fast change in depth. Rock points, gravel bars, rip-rap, boulder fields. If your jig comes back scuffed, you are in the right place.}
\telosfact{Season}{Shallow and aggressive in May--June, deep and moody in July--August, then shallow and violent again in late September and October.}
\end{whenwhere}

\section{How}
\begin{telosfacts}[0.17]
\telosfact{Best all-round}{A tube jig dragged and hopped on the bottom. It looks exactly like a fleeing crayfish and nothing else is needed most days.}
\telosfact{Deep and finicky}{Drop-shot with a small minnow-shaped plastic, \sheetref{drop-shot}. This is how you fish Pine Lake's east wall in August.}
\telosfact{Cold water}{A suspending jerkbait, worked with long pauses. In 45--55\,\unitof{^\circ F} water the pause is the whole presentation.}
\telosfact{Live bait}{A live crayfish or a large leech on a plain hook with one split shot, \sheetref{split shot}. Deadly and simple.}
\end{telosfacts}

\begin{telospanel}{The one thing to remember}
Smallmouth live where the bottom is hard. Largemouth live where the cover is soft. On the same lake, on the same day, that one distinction sorts almost every spot.
\end{telospanel}

\section{Worth knowing}
\begin{telosfacts}[0.17]
\telosfact{Slower than you think}{Smallmouth grow more slowly than largemouth in the same water. A 17\,in smallmouth in southern Wisconsin is commonly eight to ten years old. There are not many of them and they do not replace quickly.}
\telosfact{They hunt in groups}{Unlike largemouth, smallmouth commonly patrol in loose packs of similar-sized fish. If you catch one on a bar, cast straight back --- the school is usually still there, and the competition makes them bolder.}
\telosfact{Crayfish, overwhelmingly}{Stomach studies on rocky lakes routinely find crayfish as the dominant food by volume. Everything about a tube jig --- the profile, the bottom contact, the puff of sediment --- is an imitation of one fleeing backwards.}
\telosfact{They fight above their weight}{A smallmouth's caudal peduncle --- the muscle in front of the tail --- is proportionally larger than a largemouth's, and its habit of running to open water rather than into cover is why a two-pounder feels like four.}
\telosfact{The nest is guarded hard}{A male smallmouth will strike a lure dropped on his bed out of aggression, not hunger, and he will keep doing it. It is easy fishing and it costs the year class. Move off beds you can see.}
\end{telosfacts}
