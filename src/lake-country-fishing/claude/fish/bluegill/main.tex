\telosmasthead[Lepomis macrochirus]
\begin{speciestop}{bluegill}{\dentoncredit{Blue Gill Sun Fish}}
\telosfact{Tell it apart}{A \textbf{solid black ear flap} with no coloured margin, a dark smudge at the rear base of the soft dorsal fin, a small mouth, and long pointed pectoral fins. Compare with the pumpkinseed, which has an orange-red spot on the flap and wavy blue lines on the cheek.}
\telosfact{Size here}{6--8\,in is a good fish; a 9\,in bluegill is a genuine trophy and worth releasing so it keeps breeding.}
\telosfact{Spawn}{67--70\,\unitof{^\circ F}, late May into June, in dense colonies of saucer-shaped beds on sand or gravel in 1--4 ft. You can see the beds from the boat and often smell them.}
\telosfact{Eats}{Insect larvae, small crustaceans, snails. A small mouth means small baits.}
\telosfact{Season}{\telosbarrow{\telosseasonbar{3,3,2,3,5,5,4,4,4,3,2,3}}}
\telosfact{Time of day}{\telosbarrow{\telosdaybar{3,4,5,1}}}
\end{speciestop}

\section{Where}
\begin{whenwhere}
\telosfact{Pine}{Spots \textbf{6} (west bay) and \textbf{1} on the beds in June, then out to the deep weed edge at \textbf{5} --- 12--18 ft --- for the rest of the summer. Bluegill on Pine go deeper than most people expect.}
\telosfact{North}{Spots \textbf{1} and \textbf{7}, and the weed flats generally. 4--10 ft.}
\telosfact{Winter}{Basin edges and remaining green weed at 10--20 ft. Bays \textbf{6} on Pine, \textbf{1} on North.}
\end{whenwhere}

\section{How}
\begin{telosfacts}[0.17]
\telosfact{The rig}{A slip bobber, a small split shot, and a size 8 or 10 hook, \sheetref{slip bobber}. Set the depth so the bait rides just above the weed tops.}
\telosfact{Bait}{A piece of nightcrawler --- not a whole one --- a waxworm, a leaf worm, or a small leech. Bluegill have small mouths and will strip a big bait without ever getting hooked.}
\telosfact{Line}{4\,lb. Anything heavier telegraphs badly on a small float and costs you bites.}
\telosfact{With a child}{This is the fish. Anchor over a bed in June, use a fixed bobber a foot or two deep, and expect constant action. Barbless or pinched barbs make unhooking fast and safe.}
\end{telosfacts}

\begin{telospanel}{Fish the beds gently}
Bedding bluegill are easy to catch and easy to strip out of a bay. Take a meal, not a cooler. The biggest males guard the beds, and removing them all is how a lake ends up full of stunted four-inch fish.
\end{telospanel}

\section{Worth knowing}
\begin{telosfacts}[0.17]
\telosfact{There are two kinds of male}{Large \emph{parental} males build and guard the nests. Smaller \emph{sneaker} males never build one --- they mature years earlier, stay small, and dart into another male's nest to fertilise eggs. The two strategies coexist permanently in every bluegill population.}
\telosfact{Which is why big ones matter}{Remove the large parental males from a bay and the population tilts toward small early-maturing fish. That is the mechanism behind a lake full of stunted four-inch bluegill --- it is a consequence of harvest, not of overcrowding alone.}
\telosfact{Colonial nesting}{Beds are built in dense colonies, sometimes hundreds of saucers touching edge to edge in a few feet of water. You can see them from the boat, and on a calm June evening you can smell them.}
\telosfact{Age}{A 7\,in bluegill in these lakes is commonly five to seven years old. A 9\,in one may be ten. They are far older than their size suggests.}
\telosfact{The pectoral fin is a giveaway}{Long and pointed on a bluegill, short and rounded on a pumpkinseed. Useful when the ear flap is ambiguous on a hybrid.}
\end{telosfacts}

\speciesfieldproof{Is the ear flap solid black, the mouth small, and the pectoral fin long and pointed?}{Pumpkinseed show an orange-red flap margin; hybrids may mix marks, so do not decide from body colour.}{flatten the tail naturally and measure to its longest point.}{Ear flap, mouth, and pectoral shape should all agree.}
