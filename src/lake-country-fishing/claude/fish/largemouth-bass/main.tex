\telosmasthead[Micropterus salmoides]
\begin{speciestop}{largemouth-bass}{\dentoncredit{The Large-Mouthed Black Bass}}
\telosfact{Tell it apart}{The jaw. On a largemouth the upper jaw reaches \emph{past} the back of the eye; on a smallmouth it stops under the eye. Largemouth are green with one ragged dark stripe down the side, and the dorsal fin is pinched almost in two.}
\telosfact{Size here}{12--16\,in is a normal fish, 18\,in is a good one, 20\,in is a trophy for these lakes. State record 11\,lb 3\,oz.}
\telosfact{Spawn}{Beds in 1--4 ft of sand or gravel in protected bays when the water holds 60--65\,\unitof{^\circ F} --- late May into June. The male fans the bed and guards the fry for weeks afterward.}
\telosfact{Eats}{Anything it can fit: bluegill, perch, crayfish, frogs, young of its own kind.}

\telosfact{Time of day}{\telosbarrow{\telosdaybar{5,2,5,3}}}
\end{speciestop}

\section{Where}
\begin{whenwhere}
\telosfact{Pine}{Spots \textbf{5} (west shelf), \textbf{6} (west bay), \textbf{7} (south-east bay). The water is so clear that the weed line runs to 15--20 ft, and so does the bass. Do not stop fishing at 8 ft because that is where you would stop on a normal lake.}
\telosfact{North}{Spots \textbf{1}, \textbf{4}, \textbf{5}, \textbf{7}. Stained water pulls them shallow --- 4--10 ft is the band.}
\telosfact{Structure}{Weed edges, isolated weed clumps, fallen wood, docks, and any hard-to-soft bottom transition. A largemouth wants something to hide behind and open water to charge into.}
\end{whenwhere}

\section{How}
\begin{telosfacts}[0.17]
\telosfact{In the weeds}{Texas-rigged plastic worm or creature bait, \sheetref{Texas rig}. Weedless, sinks slowly, comes through everything.}
\telosfact{Pressured or clear}{Weightless wacky-rigged worm, \sheetref{wacky rig}. On Pine Lake in bright sun this out-fishes everything else.}
\telosfact{Deep}{A jig-and-craw or a Carolina rig on the outside weed edge, \sheetref{Carolina rig}.}
\telosfact{First and last light}{A walking topwater or a popper over the weed tops. The best half hour of the day.}
\telosfact{For a beginner}{A nightcrawler under a slip bobber set to hang just above the weeds. It is not fancy and it works.}
\end{telosfacts}

\begin{telospanel}{The one thing to remember}
Largemouth relate to \emph{cover}, not depth. Find the edge --- of the weeds, of the shade, of the drop --- and fish it. In the middle of a featureless flat there is no bass.
\end{telospanel}

\section{Worth knowing}
\begin{telosfacts}[0.17]
\telosfact{How old is it?}{In water this cold, a largemouth reaches about 12\,in in four years, 15\,in in six, and 18\,in in eight to ten. The 20\,in fish you are hoping for is older than the boy holding the rod.}
\telosfact{It lives in a small world}{Tagging studies consistently show largemouth holding home ranges of only a few acres, often returning to the same log or dock edge for years. A good spot is good repeatedly --- which is why the numbered spots on the lake sheets are worth learning.}
\telosfact{It hears with its body}{The lateral line detects pressure waves at a few metres --- a bass finds a lure in muddy water or full dark by feel. This is why a thumping spinnerbait blade works at night and a silent finesse worm does not.}
\telosfact{It eats by suction}{That bucket mouth opens fast and the fish inhales a volume of water. It is why a slack-line take feels like nothing, and why fixed-position baits like a wacky worm get eaten so confidently.}
\telosfact{Cold shuts it down}{Below about \qty{50}{^\circ F} a largemouth's metabolism drops so far it may feed once every several days. That is not a fish being fussy --- it genuinely does not need your lure.}
\end{telosfacts}
