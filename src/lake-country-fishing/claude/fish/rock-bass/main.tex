\telosmasthead[Ambloplites rupestris]
\begin{speciestop}{rock-bass}{\dentoncredit{Rock Bass}}
\telosfact{Tell it apart}{A bright \textbf{red eye}, a large mouth for a panfish, and rows of dark spots along the scale lines giving a dotted-stripe look. The clinching check: \textbf{six spines in the anal fin}. Bluegill, pumpkinseed and the other sunfish have three.}
\telosfact{Colour}{It can flush from pale olive to near-black in seconds, so colour alone will mislead you. Use the eye and the anal spines.}
\telosfact{Size here}{6--9\,in. Abundant on Pine Lake, whose gravel bottom is exactly what they want.}
\telosfact{Spawn}{Late May into June at 60--70\,\unitof{^\circ F}, in shallow nests scooped on gravel near rock, often close to smallmouth beds.}
\telosfact{Eats}{Crayfish, insect larvae and small fish. Aggressive well out of proportion to its size.}
\telosfact{Season}{\telosbarrow{\telosseasonbar{2,2,2,3,4,5,5,5,4,3,2,2}}}
\end{speciestop}

\section{Where and how}
\begin{whenwhere}
\telosfact{Habitat}{Rock, gravel, rip-rap and boulder. Pine spots \textbf{2}, \textbf{3}, \textbf{4} and \textbf{8}; North spots \textbf{3} and \textbf{4}. Wherever you fish for smallmouth you will find rock bass first.}
\telosfact{Depth}{2--20 ft, tight to the bottom and tight to cover.}
\telosfact{Rig}{They will take anything. A split-shot rig with a worm, a small jig, a tube --- it hardly matters. \sheetref{split shot}}
\telosfact{Counts as}{Panfish, under the 25 per day aggregate limit.}
\telosfact{Eating}{Perfectly good --- white and mild, if a little softer than perch. Under-appreciated.}
\end{whenwhere}

\begin{telospanel}{Use them}
Rock bass are the fish that saves a slow morning with a child, and they are the fish that tells you your smallmouth spot has the right bottom. Neither is a small thing. If you are catching rock bass on a new piece of structure, keep working it.
\end{telospanel}

\section{Worth knowing}
\begin{telosfacts}[0.17]
\telosfact{It changes colour in seconds}{Rock bass can shift from pale olive to almost black quickly, matching the bottom they are over. Colour is therefore useless for identification --- use the red eye and the six anal spines.}
\telosfact{Six spines, not three}{Anal-fin spine count is the clean division between rock bass (six) and the true sunfish (three). It is the sort of check that settles an argument on the water in five seconds.}
\telosfact{They tell you the bottom is right}{Rock bass need hard bottom. Catching them on unfamiliar structure is direct evidence that you are over rock or gravel --- which is exactly where the smallmouth you actually wanted will be. \sheetref{Reading Water}}
\telosfact{Aggressive out of proportion}{An 8\,in rock bass will attack a lure meant for a three-pound bass. This is a nuisance when you want bass and a gift when you have a bored child in the boat.}
\telosfact{Age}{A 9\,in rock bass may be seven or eight years old --- they are slow growers and longer-lived than their abundance suggests.}
\end{telosfacts}

\speciesfieldproof{Can you count six hard spines at the front of the anal fin?}{True sunfish such as bluegill and pumpkinseed have three anal spines; red eye alone is not decisive.}{fold the dorsal safely, press the closed snout to the stop, and read the longest tail point.}{Six anal spines plus the large mouth settle the identification even when body colour changes.}
