\telosmasthead[Sander vitreus]
\begin{speciestop}{walleye}{\dentoncredit{The Pike Perch or Wall-Eyed Pike}}
\telosfact{Tell it apart}{A \textbf{white tip on the lower lobe of the tail} and a dark blotch at the rear base of the spiny dorsal fin. The eye is large and glassy --- a reflective layer behind the retina lets it see in water too dark for its prey. Long canine teeth.}
\telosfact{Not a sauger}{Sauger have rows of black spots on the spiny dorsal, dark saddles, and no white tail tip. Not expected here, but worth knowing.}
\telosfact{Size here}{15--22\,in. The 18\,in minimum means most fish you catch go back. State record 18\,lb.}
\telosfact{Spawn}{Earliest of the gamefish after pike: 42--50\,\unitof{^\circ F}, late March into April, over rock and gravel shorelines and in inflows. Pine Lake is stocked with large fingerlings in alternate years.}

\telosfact{Time of day}{\telosbarrow{\telosdaybar{5,1,5,4}}}
\end{speciestop}

\section{Where}
\begin{whenwhere}
\telosfact{Pine}{Spots \textbf{3} (east wall), \textbf{4} (the bar) and \textbf{8}. In high summer they may suspend over the basin following cisco --- do not assume they are on the bottom.}
\telosfact{North}{Spots \textbf{3}, \textbf{4} (south-end shoals) and \textbf{5} (the neck). Stained water means they will be shallower than on Pine.}
\telosfact{Depth}{The 18--30 ft break most of the year. Shallower at dawn, dusk and after dark; deeper in bright calm.}
\telosfact{Weather}{Grey and windy is a walleye day. Wind on a point breaks up the light and pushes bait against structure --- fish the windy side.}
\end{whenwhere}

\section{How}
\begin{telosfacts}[0.17]
\telosfact{Classic}{Lindy rig with a leech, half a crawler, or a small minnow, dragged slowly along the contour, \sheetref{lindy rig}. Slow means slower than you think.}
\telosfact{Cold water}{Jig-and-minnow, \sheetref{jig-and-minnow}, worked with tiny lifts. Most bites feel like weight, not a strike.}
\telosfact{Steep faces}{A three-way rig holds a precise depth where a lindy rig will not, \sheetref{three-way rig}.}
\telosfact{Timing}{The last forty minutes of light beat the rest of the day. Be on the spot before the sun goes.}
\end{telosfacts}

\begin{telospanel}{The one thing to remember}
Walleye are not hard to catch; they are hard to be present for. Their eyes give them an advantage in dim light, so they feed when you would rather be at the landing. Fish the edges of the day.
\end{telospanel}

\section{Worth knowing}
\begin{telosfacts}[0.17]
\telosfact{The eye is the whole fish}{The \emph{tapetum lucidum} is a reflective layer behind the retina that bounces light back through it a second time. It gives a walleye roughly a full order of magnitude of light advantage over the perch it hunts --- and it is why the fish avoids bright sun, not because it dislikes it but because in daylight the advantage disappears.}
\telosfact{So fish the light, not the clock}{Dawn, dusk, night, a chop on the water, a cloudy front, stained water: all the same thing to a walleye. On Pine Lake's very clear water the effect is exaggerated, which is why the bite window there is narrow and worth being present for.}
\telosfact{Growth}{Females grow considerably faster and larger than males. Nearly every walleye over 24\,in is a female. In this part of the state an 18\,in fish is roughly four to six years old.}
\telosfact{Stocked, not self-sustaining}{Pine Lake receives large fingerling walleye in alternate years. That is a strong hint that natural reproduction is limited here --- the population is a managed one, and the size structure reflects stocking years.}
\telosfact{They travel at night}{Radio-tracking on similar lakes shows walleye moving substantial distances after dark, up onto flats they never visit in daylight. The fish that was not on your break at noon may be in 8 ft of water at eleven at night.}
\end{telosfacts}
