\telosmeta{Lake Country Fishing}{Muskellunge}{An incidental fish here --- but know it on sight}{2026-07-25}
\begin{document}
\telosmasthead[Esox masquinongy]
\begin{speciestop}{muskellunge}{\dentoncredit{The Mascalonge}}
\telosfact{Pattern}{Dark bars or spots on a \emph{light} body --- the exact reverse of a northern pike.}
\telosfact{Cheek}{Lower half of \emph{both} the cheek and the gill cover is scaleless.}
\telosfact{Jaw pores}{Six to nine per side. A pike has five or fewer.}
\telosfact{Tail}{Lobes come to sharper points than a pike's.}
\telosfact{Minimum}{40\,in, 1 per day, open water only.}
\end{speciestop}

\begin{telospanel}{Why this sheet exists}
Muskellunge are \textbf{not} among the species the DNR lists as common in Pine Lake or North Lake. They are nonetheless regulated on both waters, at a 40\,in minimum and one fish per day. If you hook a large esocid here it is almost certainly a northern pike --- but you must be able to prove that to yourself before you decide what to do with it, because the two fish carry different minimum lengths and different seasons.
\end{telospanel}

\section{Musky or pike? Three checks, in order}
\begin{telosfacts}[0.17]
\telosfact{1. Pattern}{\textbf{Musky:} dark bars or spots on a light background. \textbf{Pike:} light bean-shaped spots on a dark background. The two are photographic negatives of one another.}
\telosfact{2. Cheek \& gill cover}{Look at the side of the head. \textbf{Musky:} the lower half of \emph{both} the cheek and the gill cover is scaleless. \textbf{Pike:} the whole cheek is scaled; only the lower gill cover is bare.}
\telosfact{3. Jaw pores}{Turn the head and count the pores along one side of the underside of the lower jaw. \textbf{Musky:} six to nine. \textbf{Pike:} five or fewer. This is the check that settles an argument.}
\telosfact{Tail}{A musky's tail lobes come to sharper points; a pike's are more rounded. Useful as confirmation, not as the deciding test.}
\telosfact{Tiger musky}{A pike-musky hybrid, with bold wavy bars and rounded tail lobes. It falls under the musky regulations.}
\end{telosfacts}

\section{If you do hook one}
\begin{whenwhere}
\telosfact{Decide fast}{Measure it in the net, in the water. Under 40\,in it goes back, full stop.}
\telosfact{Gear}{A big musky cannot be landed responsibly on panfish tackle. A large net, long pliers, jaw spreaders and bolt cutters for a buried hook are the minimum.}
\telosfact{Release}{Keep it horizontal and in the water. A musky is a long, heavy fish and its own weight in air injures it. Revive it facing into the current or move it gently forward until it swims off under its own power.}
\telosfact{Season}{Open water only, 2 May--31 December --- unlike pike, there is no ice season for musky.}
\end{whenwhere}

\regsstrip{40\,in minimum, 1 per day, open water only, 2 May--31 December.  A misidentified fish is still a violation.}
