\telosmasthead[Perca flavescens]
\begin{speciestop}{yellow-perch}{\dentoncredit{Yellow or Barred Perch}}
\telosfact{Tell it apart}{Six to eight dark \textbf{vertical bars} over a yellow-green body, with orange lower fins. Two separate dorsal fins, the first spiny. Unmistakable.}
\telosfact{Size here}{6--10\,in. A 10\,in perch is a fine fish and among the best eating in either lake.}
\telosfact{Spawn}{45--52\,\unitof{^\circ F}, April, and unlike any other fish here the female lays a long accordion-folded ribbon of eggs draped over weeds and brush. You will see them in the shallows.}
\telosfact{Eats}{Insect larvae, small crustaceans, minnows and fish eggs. Perch are also the primary food of walleye and pike --- which is exactly why they school.}
\telosfact{Habit}{\textbf{A daytime feeder.} Perch shut down at night, when walleye switch on. They travel in tight schools on the bottom.}
\telosfact{Season}{\telosbarrow{\telosseasonbar{5,5,3,3,3,3,3,3,4,4,4,5}}}
\telosfact{Time of day}{\telosbarrow{\telosdaybar{3,5,3,0}}}
\end{speciestop}

\section{Where}
\begin{whenwhere}
\telosfact{Pine}{The deep weed edge at spot \textbf{5}, the flats around the base of the bar \textbf{4}, and the 15--30 ft bottom generally.}
\telosfact{North}{Spots \textbf{1}, \textbf{3} and \textbf{4}. Shallower than on Pine because of the stain.}
\telosfact{Bottom}{Perch stay on or near the bottom on clean sand, gravel or the base of weeds. If you are more than a foot off the bottom you are fishing over them.}
\telosfact{Winter}{One of the best ice fish in either lake, on the same 15--30 ft flats.}
\end{whenwhere}

\section{How}
\begin{telosfacts}[0.17]
\telosfact{Best rig}{A small jig tipped with a minnow head, a piece of crawler or a waxworm, fished right on the bottom. Or a drop-shot with a very short tag, \sheetref{drop-shot}.}
\telosfact{Move to find them}{Perch schools are tight. If you catch two and then nothing, the school moved --- move with it rather than waiting.}
\telosfact{Stay on them}{When you find a school, drop straight back down immediately. A hooked fish drags the school with it and a slow re-bait costs you the whole group.}
\end{telosfacts}

\begin{telospanel}{The lesson to teach}
Perch and walleye eat the same food in the same water but never at the same hour: one has eyes built for daylight, the other for dusk. That single fact explains most of the daily rhythm of both lakes.
\end{telospanel}

\section{Worth knowing}
\begin{telosfacts}[0.17]
\telosfact{Schools sort themselves by size}{Perch shoal by year class, so the fish you catch in one spot tend to be the same length. If they are all too small, move --- you are not going to find a big one in that school, because it is not in it.}
\telosfact{The egg ribbon}{Female perch lay a single accordion-folded gelatinous ribbon, sometimes two metres long, draped over weeds and brush. Nothing else in these lakes does this, and in April you can find them in the shallows.}
\telosfact{Daylight only, and that is unusual}{Perch have poor low-light vision and essentially stop feeding at night --- the exact inverse of walleye, which hunt them. Predator and prey partition the same water by the clock.}
\telosfact{They are the currency of the lake}{Perch are the primary forage for walleye and pike here. A good perch year is followed by good walleye growth, and the size of the perch you catch tells you what the predators are eating.}
\telosfact{Age}{A 10\,in perch is commonly five to seven years old and is a genuinely good fish anywhere in Wisconsin.}
\end{telosfacts}
