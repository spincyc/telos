\telosmasthead[Pomoxis nigromaculatus]
\begin{speciestop}{black-crappie}{\dentoncredit{Calico Bass; Strawberry Bass (Pomoxis sparoides)}}
\telosfact{Tell it apart}{\textbf{Count the dorsal spines: seven or eight} on a black crappie, five or six on a white crappie. The speckling is scattered and irregular, not organised into vertical bars. The body is deep, thin and silvery.}
\telosfact{The paper mouth}{The membrane around the jaw tears easily. Do not set hard and do not lift a crappie by the line --- this is the fish people lose at the side of the boat.}
\telosfact{Size here}{8--11\,in. A 12\,in crappie is a very good fish.}
\telosfact{Spawn}{58--64\,\unitof{^\circ F}, May, in 3--8 ft near brush, wood or the inside weed edge. Slightly earlier and slightly deeper than bluegill.}
\telosfact{Eats}{Small minnows and zooplankton. More fish-eating than the sunfish, which is why a minnow beats a worm.}
\telosfact{Season}{\telosbarrow{\telosseasonbar{4,4,3,4,5,3,2,2,3,4,3,4}}}
\telosfact{Time of day}{\telosbarrow{\telosdaybar{4,1,5,3}}}
\end{speciestop}

\section{Where}
\begin{whenwhere}
\telosfact{Pine}{Spot \textbf{7} (south-east bay) and \textbf{6} in spring; the edges of the mid-lake bar \textbf{4} in summer, where they suspend off the drop.}
\telosfact{North}{Spots \textbf{1} and \textbf{4}, and anywhere with standing wood.}
\telosfact{The key habit}{Crappie \textbf{suspend}. They roam in schools at a chosen depth over deep water, ignoring the bottom entirely. Finding the depth matters far more than finding the spot.}
\end{whenwhere}

\section{How}
\begin{telosfacts}[0.17]
\telosfact{Find the depth}{Cast a small jig, count it down --- five, ten, fifteen --- and retrieve. When you get bit, remember the count and repeat it exactly.}
\telosfact{Best bait}{A small live minnow under a slip bobber, \sheetref{slip bobber}, set precisely at the depth you found. Crappie feed \emph{upward}; set the bait slightly above them, never below.}
\telosfact{Artificial}{1/16 or 1/32\,oz jig with a soft plastic tail, in white or chartreuse. Barely move it.}
\telosfact{Hookset}{A steady sweep, not a snap. Paper mouth.}
\telosfact{When}{Dusk is the crappie hour. They also feed well in the first hour of an overcast morning and through the ice in January.}
\end{telosfacts}

\begin{telospanel}{The one thing to remember}
If you are catching nothing, change your \emph{depth} before you change your bait. A school two feet below your jig will not come up for it, and a school two feet above it never sees it.
\end{telospanel}
