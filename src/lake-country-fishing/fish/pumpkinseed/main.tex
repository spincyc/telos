\telosmasthead[Lepomis gibbosus]
\begin{speciestop}{pumpkinseed}{\dentoncredit{Sunfish (Eupomotis gibbosus)}}
\telosfact{Tell it apart}{An \textbf{orange-red spot on the trailing edge of the black ear flap}, and wavy pale-blue lines radiating back across the cheek and gill cover. The fins are speckled. There is \emph{no} dark blotch at the back of the soft dorsal --- that blotch belongs to the bluegill.}
\telosfact{Shape}{Rounder and deeper-bodied than a bluegill, with a slightly larger mouth and shorter, more rounded pectoral fins.}
\telosfact{Size here}{5--7\,in. Rarely as large as a good bluegill, and usually a smaller share of the catch.}
\telosfact{Spawn}{Same window as bluegill, 65--70\,\unitof{^\circ F}, often on the same colonial beds --- and the two species hybridize, which is why you will occasionally land something that fits neither description.}
\telosfact{Eats}{Snails above all --- it has crushing teeth in its throat for them --- plus insect larvae and small crustaceans.}
\telosfact{Season}{\telosbarrow{\telosseasonbar{2,2,2,3,5,5,4,4,3,3,2,2}}}
\end{speciestop}

\section{Where and how}
\begin{whenwhere}
\telosfact{Habitat}{Shallower and weedier than bluegill, and more tied to cover. Pine spots \textbf{6} and \textbf{1}; North spots \textbf{1} and \textbf{7}.}
\telosfact{Depth}{1--8 ft, in and against vegetation. If you are catching bluegill on the deep weed edge and pumpkinseed on the inside edge, that is normal.}
\telosfact{Rig}{Identical to bluegill --- slip bobber, split shot, size 8 or 10 hook, small piece of worm. \sheetref{slip bobber}}
\telosfact{Counts as}{Panfish, under the 25 per day aggregate limit.}
\end{whenwhere}

\begin{telospanel}{Worth knowing on the bed}
When you find a June sunfish colony you will usually be catching both species plus the occasional hybrid. Sorting them on the spot is a good five-minute lesson for a child: flip the gill cover into the light and look for the orange dot.
\end{telospanel}
