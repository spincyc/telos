\telosmasthead[Esox lucius]
\begin{speciestop}{northern-pike}{\dentoncredit{The Pike}}
\telosfact{Tell it apart}{\textbf{Light bean-shaped spots on a dark green body} --- light-on-dark. The cheek is fully scaled and the lower half of the gill cover is bare. Five sensory pores along each side of the lower jaw.}
\telosfact{Not a musky}{A musky is the photographic negative: \emph{dark} marks on a \emph{light} body, the lower half of both the cheek and gill cover bare, and six to nine pores per side. At a 40\,in minimum on musky and 26\,in on pike, this matters.}
\telosfact{Size here}{20--30\,in is the working range; 36\,in and up is a real fish. State record 38\,lb.}
\telosfact{Spawn}{The first gamefish to go, immediately after ice-out at 40--45\,\unitof{^\circ F}, in flooded marsh grass and shallow vegetation --- often before the ice is fully gone from the main lake.}
\telosfact{Season}{\telosbarrow{\telosseasonbar{3,3,3,5,4,3,2,2,3,5,5,3}}}
\telosfact{Time of day}{\telosbarrow{\telosdaybar{4,3,5,1}}}
\end{speciestop}

\section{Where}
\begin{whenwhere}
\telosfact{Pine}{Spots \textbf{1} and \textbf{9} in spring and fall, \textbf{5} along the deep weed edge in summer. In a lake this cold and clear, the largest pike go deep and cool in July and August --- 25--40 ft on the edge of the basin, not in the shallows.}
\telosfact{North}{Spots \textbf{2} (outlet), \textbf{5} (the neck) and \textbf{6} (river inlet). Current and weed. This lake fishes more like a classic pike water than Pine does.}
\telosfact{Cover}{Weed edges, points, and any narrows. Pike are ambush hunters that want a wall at their back and open water in front.}
\end{whenwhere}

\section{How}
\begin{telosfacts}[0.17]
\telosfact{Cover water}{A large spinnerbait or a spoon over and along the weed tops. Fast, snag-resistant, and it finds the aggressive fish.}
\telosfact{Big fish}{A large sucker on a quick-strike rig under a float, \sheetref{quick-strike rig}. This is how most of the biggest pike in these lakes are taken.}
\telosfact{Ice}{Tip-ups with a live shiner set just off the bottom along the weed edge, \sheetref{tip-up}.}
\telosfact{Leader}{\textbf{Always.} A wire or heavy fluorocarbon leader. Pike teeth cut straight line, and losing a fish with a hook in it is a bad outcome for the fish, not just for you.}
\end{telosfacts}

\begin{teloshazard}{Handling a pike}
The teeth are backward-facing and will cut you to the bone. Long pliers and jaw
spreaders, always; fingers never go in the mouth or under the gill plate.
Support a big fish horizontally with a wet hand under the belly. Above roughly
\qty{70}{^\circ F} a released pike is fragile --- unhook it in the net, in the
water, and let it go.
\end{teloshazard}
